using System.Collections;
using System.Collections.Generic;
using UnityEngine;

public class firstPersonController : MonoBehaviour
{
    public float walkSpeed = 5f;
    public float sensitivity = 2f;
    public float gravity = -20f;

    public float sprintSpeed = 8f;

    public float maxStamina = 100f;
    public float currStamina = 100f;
    public float staminaDrain = 25f;
    public float staminaRegen = 15f;

    public float crouchHeight = 1f;
    public float standingHeight = 2f;
    public float crouchSpeed = 2.5f;

    public float cameraStandingHeight = 1.7f;
    public float cameraCrouchingHeight = 0.9f;

    // Phone camera sensitivity
    public float phoneLookSensitivity = 100f;

    private bool isCrouching = false;

    public UnityEngine.UI.Slider staminaBar;

    public Transform playerCamera;

    private CharacterController charControl;

    private float vertiVelo;
    private float cameraPitch = 0f;

    // Phone receiver
    private PhoneReceiver phone;


    void Start()
    {
        charControl = GetComponent<CharacterController>();

        phone = FindObjectOfType<PhoneReceiver>();

        Cursor.lockState = CursorLockMode.Locked;
        Cursor.visible = false;
    }


    void Update()
    {
        // =====================================================
        // CAMERA LOOK
        // =====================================================

        // Keyboard / mouse
        float mouseX = Input.GetAxis("Mouse X") * sensitivity;
        float mouseY = Input.GetAxis("Mouse Y") * sensitivity;

        // Phone right joystick
        float phoneX = 0f;
        float phoneY = 0f;

        if (phone != null)
        {
            phoneX =
                phone.rightX *
                phoneLookSensitivity *
                Time.deltaTime;

            phoneY =
                phone.rightY *
                phoneLookSensitivity *
                Time.deltaTime;
        }

        // Combine mouse + phone
        float lookX = mouseX + phoneX;
        float lookY = mouseY + phoneY;

        // Horizontal rotation
        transform.Rotate(Vector3.up * lookX);

        // Vertical camera rotation
        cameraPitch -= lookY;

        cameraPitch =
            Mathf.Clamp(
                cameraPitch,
                -90f,
                90f
            );

        playerCamera.localRotation =
            Quaternion.Euler(
                cameraPitch,
                0f,
                0f
            );


        // =====================================================
        // MOVEMENT
        // =====================================================

        // Keyboard
        float keyboardX =
            Input.GetAxis("Horizontal");

        float keyboardZ =
            Input.GetAxis("Vertical");

        // Phone left joystick
        float phoneMoveX = 0f;
        float phoneMoveZ = 0f;

        if (phone != null)
        {
            phoneMoveX = phone.leftX;
            phoneMoveZ = phone.leftY;
        }

        // Combine keyboard + phone
        float x =
            keyboardX + phoneMoveX;

        float z =
            keyboardZ + phoneMoveZ;

        // Prevent diagonal speed from becoming too fast
        Vector2 inputVector =
            new Vector2(x, z);

        inputVector =
            Vector2.ClampMagnitude(
                inputVector,
                1f
            );

        x = inputVector.x;
        z = inputVector.y;

        Vector3 move =
            transform.right * x +
            transform.forward * z;


        // =====================================================
        // CROUCH
        // =====================================================

        bool keyboardCrouch =
            Input.GetKey(KeyCode.LeftControl);

        bool phoneCrouch =
            phone != null &&
            phone.crouch;

        isCrouching =
            keyboardCrouch ||
            phoneCrouch;


        // =====================================================
        // SPEED
        // =====================================================

        float currentSpeed = walkSpeed;

        if (isCrouching)
        {
            currentSpeed = crouchSpeed;
        }


        // =====================================================
        // SPRINT
        // =====================================================

        bool keyboardSprint =
            Input.GetKey(KeyCode.LeftShift);

        bool phoneSprint =
            phone != null &&
            phone.sprint;

        bool sprinting =
            keyboardSprint ||
            phoneSprint;

        // Don't sprint while crouching
        if (sprinting &&
            !isCrouching &&
            move.magnitude > 0.1f &&
            currStamina > 0f)
        {
            currentSpeed = sprintSpeed;

            currStamina -=
                staminaDrain *
                Time.deltaTime;
        }
        else
        {
            currStamina +=
                staminaRegen *
                Time.deltaTime;
        }


        // Keep stamina within limits
        currStamina =
            Mathf.Clamp(
                currStamina,
                0f,
                maxStamina
            );


        // Update stamina bar
        if (staminaBar != null)
        {
            staminaBar.value =
                currStamina;
        }


        // =====================================================
        // MOVE PLAYER
        // =====================================================

        charControl.Move(
            move *
            currentSpeed *
            Time.deltaTime
        );


        // =====================================================
        // CAMERA HEIGHT
        // =====================================================

        float targetCameraHeight;

        if (isCrouching)
        {
            targetCameraHeight =
                cameraCrouchingHeight;
        }
        else
        {
            targetCameraHeight =
                cameraStandingHeight;
        }

        Vector3 cameraPosition =
            playerCamera.localPosition;

        cameraPosition.y =
            Mathf.Lerp(
                cameraPosition.y,
                targetCameraHeight,
                10f * Time.deltaTime
            );


        // =====================================================
        // PLAYER HEIGHT
        // =====================================================

        float targetHeight;

        if (isCrouching)
        {
            targetHeight =
                crouchHeight;
        }
        else
        {
            targetHeight =
                standingHeight;
        }

        charControl.height =
            Mathf.Lerp(
                charControl.height,
                targetHeight,
                10f * Time.deltaTime
            );

        playerCamera.localPosition =
            cameraPosition;


        // =====================================================
        // GRAVITY
        // =====================================================

        if (charControl.isGrounded &&
            vertiVelo < 0)
        {
            vertiVelo = -2f;
        }

        vertiVelo +=
            gravity *
            Time.deltaTime;

        charControl.Move(
            Vector3.up *
            vertiVelo *
            Time.deltaTime
        );
    }
}
